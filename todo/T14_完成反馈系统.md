### 3.1 卡片抽取逻辑 `utils/card-draw.js`

```javascript
/**
 * 卡片抽取核心逻辑
 */

/**
 * 抽取今日卡片
 * @param {Array} deck 卡组(3张卡)
 * @param {number} streak 当前连续天数
 * @returns {Object} 今日卡片
 */
function drawTodayCard(deck, streak) {
  // 确保卡组有效
  if (!deck || deck.length === 0) {
    return getDefaultCard();
  }
  
  const baseCard = deck.find(c => c.type === 'base');
  const mainCard = deck.find(c => c.type === 'main');
  const bonusCard = deck.find(c => c.type === 'bonus');
  
  // 连续天数不足2天: 必定保底卡
  if (streak < 2) {
    return baseCard || deck[0];
  }
  
  // 连续2天+: 按概率抽取
  const rand = Math.random();
  
  if (rand < 0.1 && bonusCard) {
    // 10%: 彩蛋卡
    return bonusCard;
  } else if (rand < 0.4 && mainCard) {
    // 30%: 主线卡
    return mainCard;
  } else {
    // 60%: 保底卡
    return baseCard || deck[0];
  }
}

/**
 * 获取救援卡
 * @param {Array} deck 卡组
 * @returns {Object} 救援卡(保底卡或默认卡)
 */
function getRescueCard(deck) {
  // 优先从用户卡组取保底卡
  const baseCard = deck.find(c => c.type === 'base');
  if (baseCard) {
    return {
      ...baseCard,
      isRescue: true
    };
  }
  
  // 否则返回默认救援卡
  return getDefaultCard(true);
}

/**
 * 获取默认卡片
 * @param {boolean} isRescue 是否是救援卡
 * @returns {Object} 默认卡片
 */
function getDefaultCard(isRescue = false) {
  return {
    id: 'default_card',
    name: '深呼吸3次',
    desc: '✨ → 深呼吸3次',
    seconds: 15,
    difficulty: 1,
    type: 'base',
    anchor: '',
    anchorEmoji: '✨',
    wish: '默认',
    isRescue
  };
}

/**
 * 概率验证函数(用于测试)
 * @param {number} iterations 迭代次数
 * @returns {Object} 概率统计结果
 */
function verifyProbability(iterations = 10000) {
  const mockDeck = [
    { type: 'base', name: '保底' },
    { type: 'main', name: '主线' },
    { type: 'bonus', name: '彩蛋' }
  ];
  
  const counts = { base: 0, main: 0, bonus: 0 };
  
  for (let i = 0; i < iterations; i++) {
    const card = drawTodayCard(mockDeck, 5); // streak >= 2
    counts[card.type]++;
  }
  
  return {
    base: (counts.base / iterations * 100).toFixed(1) + '%',
    main: (counts.main / iterations * 100).toFixed(1) + '%',
    bonus: (counts.bonus / iterations * 100).toFixed(1) + '%'
  };
}

module.exports = {
  drawTodayCard,
  getRescueCard,
  getDefaultCard,
  verifyProbability
};
```

### 3.3 完成流程逻辑 `utils/complete-habit.js`

```javascript
/**
 * 完成微行动的完整流程
 */

const { loadState, saveState } = require('./state.js');
const { updateStreak } = require('./streak.js');
const { isToday, getTodayString } = require('./date.js');

/**
 * 完成微行动
 * @param {string} cardId 完成的卡片ID
 * @param {Object} card 卡片对象
 * @returns {Object} 完成结果
 */
function completeHabit(cardId, card) {
  const state = loadState();
  const today = getTodayString();
  
  // 1. 幂等性检查
  if (state.todayDone && isToday(state.lastDoneDate)) {
    return { 
      success: false, 
      message: '今日已完成',
      alreadyDone: true
    };
  }
  
  // 2. 更新Streak
  const newStreak = updateStreak(state);
  
  // 3. 更新Day进度
  const isDay7 = state.day >= 7;
  if (!isDay7) {
    state.day += 1;
  }
  
  // 4. 更新完成状态
  state.todayDone = true;
  state.lastDoneDate = today;
  
  // 5. 更新积分
  const scoreEarned = 10;
  state.totalScore = (state.totalScore || 0) + scoreEarned;
  
  // 6. 记录历史
  if (!state.history) state.history = [];
  state.history.push({
    date: today,
    cardName: card ? card.name : '未知咒语',
    seconds: card ? card.seconds : 0
  });
  
  // 保持历史记录最多7条
  if (state.history.length > 7) {
    state.history = state.history.slice(-7);
  }
  
  // 7. 更新咒语图鉴
  if (!state.spellBook) state.spellBook = [];
  if (card && !state.spellBook.includes(card.name)) {
    state.spellBook.push(card.name);
  }
  
  // 8. 保存状态
  saveState(state);
  
  return {
    success: true,
    streak: state.streak,
    bestStreak: state.bestStreak,
    day: state.day,
    totalScore: state.totalScore,
    scoreEarned,
    isDay7,
    spellBookCount: state.spellBook.length
  };
}

/**
 * 检查是否可以完成
 * @returns {Object} { canComplete, reason }
 */
function canComplete() {
  const state = loadState();
  
  if (!state.hasDeck || !state.deck || state.deck.length === 0) {
    return { canComplete: false, reason: '请先生成卡组' };
  }
  
  if (state.todayDone && isToday(state.lastDoneDate)) {
    return { canComplete: false, reason: '今日已完成' };
  }
  
  return { canComplete: true, reason: null };
}

module.exports = {
  completeHabit,
  canComplete
};
```

const { drawTodayCard, verifyProbability } = require('../utils/card-draw.js');

describe('drawTodayCard', () => {
  const mockDeck = [
    { type: 'base', name: '保底' },
    { type: 'main', name: '主线' },

    const stats = verifyProbability(10000);
    
    // 允许5%的误差
    const basePercent = parseFloat(stats.base);

    expect(basePercent).toBeGreaterThan(55);
    expect(basePercent).toBeLessThan(65);
