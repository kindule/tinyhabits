### 3.1 工具函数 `utils/workshop.js`

```javascript
/**
 * 行为工坊核心逻辑
 */

const { BEHAVIORS, SPLIT_RULES, DEFAULT_BEHAVIORS } = require('../data/behaviors.js');

/**
 * 召唤灵感 - 随机抽取行为
 * @param {string} wishId 当前愿望ID
 * @param {number} count 抽取数量
 * @returns {Array} 行为列表
 */
function summonBehaviors(wishId, count = 3) {
  // 过滤当前愿望相关的行为
  const relevantBehaviors = BEHAVIORS.filter(b => b.category === wishId);
  
  // 如果没有相关行为，使用默认行为
  const pool = relevantBehaviors.length > 0 ? relevantBehaviors : DEFAULT_BEHAVIORS;
  
  // 随机抽取
  const shuffled = shuffleArray([...pool]);
  return shuffled.slice(0, Math.min(count, shuffled.length));
}

/**
 * 行为裂变 - 将大行为拆分为小行为
 * @param {Object} behavior 要裂变的行为
 * @returns {Array} 裂变后的行为列表(3个)
 */
function splitBehavior(behavior) {
  // 检查是否有预定义的裂变规则
  if (behavior.splitTo && behavior.splitTo.length > 0) {
    const results = behavior.splitTo.map(id => 
      BEHAVIORS.find(b => b.id === id) || createMicroBehavior(id)
    );
    return results.slice(0, 3);
  }
  
  // 检查裂变规则库
  const rule = SPLIT_RULES[behavior.id];
  if (rule) {
    return rule.map(id => 
      BEHAVIORS.find(b => b.id === id) || createMicroBehavior(id)
    ).slice(0, 3);
  }
  
  // 通用裂变：生成更小的版本
  return generateGenericSplit(behavior);
}

/**
 * 生成通用裂变结果
 * @param {Object} behavior 原行为
 * @returns {Array} 裂变后的行为列表
 */
function generateGenericSplit(behavior) {
  const prefixes = ['准备', '开始', '完成一点'];
  const suffixes = ['的第一步', '一下', '一点点'];
  
  return [
    {
      id: `${behavior.id}_step1`,
      name: `${prefixes[0]}${behavior.name}`,
      seconds: Math.floor(behavior.seconds / 4),
      difficulty: 1,
      category: behavior.category
    },
    {
      id: `${behavior.id}_step2`,
      name: `${behavior.name}${suffixes[1]}`,
      seconds: Math.floor(behavior.seconds / 3),
      difficulty: 1,
      category: behavior.category
    },
    {
      id: `${behavior.id}_step3`,
      name: `${behavior.name}${suffixes[2]}`,
      seconds: Math.floor(behavior.seconds / 5),
      difficulty: 1,
      category: behavior.category
    }
  ];
}

/**
 * 创建微行为
 * @param {string} id 行为ID
 * @returns {Object} 行为对象
 */
function createMicroBehavior(id) {
  return {
    id,
    name: id.replace(/_/g, ' '),
    seconds: 10,
    difficulty: 1,
    category: 'general'
  };
}

/**
 * 行为合成 - 将两个行为合成新咒语
 * @param {Object} behaviorA 行为A
 * @param {Object} behaviorB 行为B
 * @returns {Object} 合成后的新行为
 */
function mergeBehaviors(behaviorA, behaviorB) {
  const templates = [
    { format: '{A}后{B}', name: `${simplifyName(behaviorA.name)}后${simplifyName(behaviorB.name)}` },
    { format: '{A}+{B}', name: `${simplifyName(behaviorA.name)}+${simplifyName(behaviorB.name)}` },
    { format: '「{A}·{B}」', name: `「${getKeyword(behaviorA.name)}·${getKeyword(behaviorB.name)}」` }
  ];
  
  const template = templates[Math.floor(Math.random() * templates.length)];
  
  return {
    id: `merged_${Date.now()}`,
    name: template.name,
    seconds: Math.min(behaviorA.seconds + behaviorB.seconds, 120),
    difficulty: Math.max(behaviorA.difficulty, behaviorB.difficulty),
    category: behaviorA.category,
    isMerged: true,
    sourceA: behaviorA.id,
    sourceB: behaviorB.id,
    anchor: behaviorA.anchor || behaviorB.anchor,
    anchorEmoji: behaviorA.anchorEmoji || behaviorB.anchorEmoji
  };
}

/**
 * 简化行为名称
 */
function simplifyName(name) {
  return name.replace(/^(去|做|来|进行)/, '').slice(0, 4);
}

/**
 * 提取关键词
 */
function getKeyword(name) {
  // 提取2-3个字的关键词
  const cleaned = name.replace(/^(去|做|来|进行)/, '');
  return cleaned.slice(0, 2);
}

/**
 * 生成卡组
 * @param {Array} materialArea 素材区行为列表
 * @param {string} currentWish 当前愿望名称
 * @returns {Array} 3张卡的数组
 */
function generateDeck(materialArea, currentWish) {
  if (materialArea.length === 0) {
    return generateDefaultDeck(currentWish);
  }
  
  // 按难度和时长分类
  const easy = materialArea.filter(b => b.seconds <= 60 && b.difficulty === 1);
  const medium = materialArea.filter(b => b.seconds > 60 && b.seconds <= 300);
  const fun = materialArea.filter(b => b.seconds <= 30);
  
  const deck = [];
  
  // 保底卡: 最简单的
  const base = easy.length > 0 
    ? easy[Math.floor(Math.random() * easy.length)]
    : materialArea[0];
  deck.push(createCard(base, 'base', currentWish));
  
  // 主线卡: 中等难度
  const main = medium.length > 0
    ? medium[Math.floor(Math.random() * medium.length)]
    : (materialArea.length > 1 ? materialArea[1] : materialArea[0]);
  deck.push(createCard(main, 'main', currentWish));
  
  // 彩蛋卡: 趣味性
  const bonus = fun.length > 0
    ? fun[Math.floor(Math.random() * fun.length)]
    : (materialArea.length > 2 ? materialArea[2] : materialArea[0]);
  deck.push(createCard(bonus, 'bonus', currentWish));
  
  return deck;
}

/**
 * 创建卡片
 */
function createCard(behavior, type, wish) {
  return {
    id: `card_${Date.now()}_${type}`,
    name: behavior.name,
    desc: `${behavior.anchorEmoji || '✨'} ${behavior.anchor || ''} → ${behavior.name}`,
    seconds: behavior.seconds || 30,
    difficulty: type === 'base' ? 1 : type === 'main' ? 2 : 1,
    type: type,
    anchor: behavior.anchor,
    anchorEmoji: behavior.anchorEmoji,
    wish: wish,
    createdAt: new Date().toISOString()
  };
}

/**
 * 生成默认卡组
 */
function generateDefaultDeck(currentWish) {
  return [
    createCard({ 
      name: '喝一杯水', 
      seconds: 30, 
      anchor: '起床后', 
      anchorEmoji: '🌅' 
    }, 'base', currentWish),
    createCard({ 
      name: '伸展身体', 
      seconds: 120, 
      anchor: '午休后', 
      anchorEmoji: '☀️' 
    }, 'main', currentWish),
    createCard({ 
      name: '深呼吸3次', 
      seconds: 15, 
      anchor: '感到压力时', 
      anchorEmoji: '😮‍💨' 
    }, 'bonus', currentWish)
  ];
}

/**
 * 洗牌
 */
function shuffleArray(array) {
  const shuffled = [...array];
  for (let i = shuffled.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [shuffled[i], shuffled[j]] = [shuffled[j], shuffled[i]];
  }
  return shuffled;
}

module.exports = {
  summonBehaviors,
  splitBehavior,
  mergeBehaviors,
  generateDeck,
  createCard,
  generateDefaultDeck
};
```

### 3.2 单元测试 `test/workshop.test.js`

```javascript
const {
  summonBehaviors,
  splitBehavior,
  mergeBehaviors,
  generateDeck
} = require('../utils/workshop.js');

describe('summonBehaviors', () => {
  test('should return correct number of behaviors', () => {
    const result = summonBehaviors('health', 3);
    expect(result.length).toBeLessThanOrEqual(3);
  });
  
  test('should return behaviors for valid wish', () => {
    const result = summonBehaviors('health', 3);
    expect(result.length).toBeGreaterThan(0);
  });
});

describe('splitBehavior', () => {
  test('should return 3 behaviors', () => {
    const behavior = { id: 'test', name: '测试行为', seconds: 60 };
    const result = splitBehavior(behavior);
    expect(result.length).toBe(3);
  });
  
  test('split behaviors should be smaller', () => {
    const behavior = { id: 'test', name: '测试行为', seconds: 60 };
    const result = splitBehavior(behavior);
    result.forEach(b => {
      expect(b.seconds).toBeLessThan(behavior.seconds);
    });
  });
});

describe('mergeBehaviors', () => {
  test('should create merged behavior', () => {
    const a = { id: 'a', name: '喝水', seconds: 30, difficulty: 1 };
    const b = { id: 'b', name: '伸展', seconds: 60, difficulty: 1 };
    
    const result = mergeBehaviors(a, b);
    
    expect(result.isMerged).toBe(true);
    expect(result.seconds).toBe(90);
  });
  
  test('merged seconds should not exceed 120', () => {
    const a = { id: 'a', name: '跑步', seconds: 100, difficulty: 2 };
    const b = { id: 'b', name: '游泳', seconds: 100, difficulty: 2 };
    
    const result = mergeBehaviors(a, b);
    
    expect(result.seconds).toBeLessThanOrEqual(120);
  });
});

describe('generateDeck', () => {
  test('should generate 3 cards', () => {
    const materials = [
      { id: '1', name: '行为1', seconds: 30, difficulty: 1 },
      { id: '2', name: '行为2', seconds: 120, difficulty: 2 },
      { id: '3', name: '行为3', seconds: 20, difficulty: 1 }
    ];
    
    const deck = generateDeck(materials, '健康');
    
    expect(deck.length).toBe(3);
    expect(deck.find(c => c.type === 'base')).toBeTruthy();
    expect(deck.find(c => c.type === 'main')).toBeTruthy();
    expect(deck.find(c => c.type === 'bonus')).toBeTruthy();
  });
  
  test('should generate default deck when materials empty', () => {
    const deck = generateDeck([], '健康');
    expect(deck.length).toBe(3);
  });
});
```
