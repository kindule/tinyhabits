# T03: 全局UI组件库

> **优先级**: P0 (必须实现)
> **预估工时**: 1.5天
> **负责人**: 前端开发
> **状态**: ✅ DONE

---

## 一、任务描述

创建全局可复用的UI组件库，包括对话框、卡片、按钮、气泡、标签等核心组件，所有组件使用设计令牌变量。

---

## 二、验收标准

- [ ] 所有组件样式使用设计令牌变量
- [ ] 组件支持自定义属性(slot、properties)
- [ ] 组件响应式适配不同屏幕
- [ ] 交互动效符合规范(按压缩放、弹跳等)
- [ ] 创建组件预览页验证效果

---

### 4.1 DialogueBubble 对话框组件

```javascript
// components/dialogue-bubble/index.js
Component({
  properties: {
    character: {
      type: String,
      value: 'arc', // 'arc' | 'mia' | 'rift'
    },
    text: {
      type: String,
      value: ''
    },
    animate: {
      type: Boolean,
      value: true
    }
  },
  
  data: {
    showAnimation: false
  },
  
  lifetimes: {
    attached() {
      if (this.properties.animate) {
        setTimeout(() => {
          this.setData({ showAnimation: true });
        }, 50);
      }
    }
  }
});
```

```html
<!-- components/dialogue-bubble/index.wxml -->
<view class="dialogue-bubble {{showAnimation ? 'animate-in' : ''}}">
  <view class="avatar avatar-{{character}}"></view>
  <view class="dialogue-content">
    <text class="dialogue-text">{{text}}</text>
  </view>
</view>
```

### 4.2 MainCard 主卡片组件

```javascript
// components/main-card/index.js
Component({
  properties: {
    selected: {
      type: Boolean,
      value: false
    },
    disabled: {
      type: Boolean,
      value: false
    }
  },
  
  methods: {
    onTap() {
      if (!this.properties.disabled) {
        this.triggerEvent('tap');
      }
    }
  }
});
```

```html
<!-- components/main-card/index.wxml -->
<view 
  class="main-card {{selected ? 'selected' : ''}} {{disabled ? 'disabled' : ''}}"
  bindtap="onTap"
>
  <slot></slot>
```


### 4.3 PrimaryButton 主按钮组件

```javascript
// components/primary-button/index.js
Component({
  properties: {
    text: {
      type: String,
      value: ''
    },
    disabled: {
      type: Boolean,
      value: false
    },
    loading: {
      type: Boolean,
      value: false
    }
  },
  
  data: {
    shimmer: false
  },
  
  methods: {
    onTap() {
      if (!this.properties.disabled && !this.properties.loading) {
        this.setData({ shimmer: true });
        setTimeout(() => this.setData({ shimmer: false }), 200);
        this.triggerEvent('tap');
      }
    }
  }
});
```

```html
<!-- components/primary-button/index.wxml -->
<button 
  class="btn-primary {{shimmer ? 'shimmer' : ''}} {{disabled ? 'disabled' : ''}}"
  disabled="{{disabled || loading}}"
  bindtap="onTap"
>
  <text wx:if="{{!loading}}">{{text}}</text>
  <text wx:else>加载中...</text>
</button>
```


### 4.4 BubbleChip 行为气泡组件

```javascript
// components/bubble-chip/index.js
Component({
  properties: {
    text: { type: String, value: '' },
    selected: { type: Boolean, value: false },
    delay: { type: Number, value: 0 }  // 动画延迟(ms)
  },
  
  data: {
    visible: false
  },
  
  lifetimes: {
    attached() {
      setTimeout(() => {
        this.setData({ visible: true });
      }, this.properties.delay);
    }
  },
  
  methods: {
    onTap() {
      this.triggerEvent('tap');
    }
  }
});
```

```css
/* components/bubble-chip/index.wxss */
.bubble-chip {
  display: inline-flex;
  align-items: center;
  padding: 16rpx 24rpx;
  border-radius: 999rpx;
  background: rgba(94, 200, 216, 0.18);
  position: relative;
  transition: transform 180ms ease-out;
  opacity: 0;
  transform: scale(0.8);
}

.bubble-chip.visible {
  animation: bubbleAppear 160ms ease-out forwards;
}

.bubble-chip.selected {
  background: rgba(246, 200, 95, 0.25);
  border: 2rpx solid var(--c-gold);
}

.bubble-chip:active {
  animation: bubblePop 180ms ease-out;
}

/* 高光效果 */
.bubble-chip::after {
  content: '';
  position: absolute;
  top: 2rpx;
  left: 10%;
  right: 10%;
  height: 30%;
  background: linear-gradient(180deg, rgba(255,255,255,0.5), transparent);
  border-radius: 999rpx;
}

@keyframes bubbleAppear {
  from {
    opacity: 0;
    transform: scale(0.8);
  }
  to {
    opacity: 1;
    transform: scale(1);
  }
}

@keyframes bubblePop {
  0% { transform: scale(1); }
  50% { transform: scale(1.08); }
  100% { transform: scale(1); }
}
```
