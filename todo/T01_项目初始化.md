### 3.3 app.js基础配置

```javascript
App({
  onLaunch() {
    // 检查更新
    this.checkUpdate();
  },
  
  onShow() {
    // 每次显示时刷新日期状态
    this.refreshDailyState();
  },
  
  globalData: {
    version: '1.0.0-MVP'
  },
  
  checkUpdate() {
    if (wx.canIUse('getUpdateManager')) {
      const updateManager = wx.getUpdateManager();
      updateManager.onUpdateReady(() => {
        wx.showModal({
          title: '更新提示',
          content: '新版本已准备好',
          success: (res) => {
            if (res.confirm) {
              updateManager.applyUpdate();
            }
          }
        });
      });
    }
  },
  
  refreshDailyState() {
    // 调用状态管理模块刷新每日状态
    const { refreshDailyState } = require('./utils/state.js');
    refreshDailyState();
  }
});
```
