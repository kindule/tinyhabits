### 3.1 工具函数 `utils/wish-sort.js`

```javascript
/**
 * 愿望排序核心逻辑
 * 基于两两对决的排序算法
 */

/**
 * 洗牌算法(Fisher-Yates)
 * @param {Array} array 
 * @returns {Array}
 */
function shuffleArray(array) {
  const shuffled = [...array];
  for (let i = shuffled.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [shuffled[i], shuffled[j]] = [shuffled[j], shuffled[i]];
  }
  return shuffled;
}

/**
 * 生成配对列表
 * @param {Array} wishes 愿望列表
 * @param {number} rounds 配对轮数
 * @returns {Array<Array>} 配对列表，每项为[wishA, wishB]
 */
function generateMatchups(wishes, rounds = 7) {
  const matchups = [];
  const usedPairs = new Set();
  
  // 生成所有可能的配对
  const allPairs = [];
  for (let i = 0; i < wishes.length; i++) {
    for (let j = i + 1; j < wishes.length; j++) {
      allPairs.push([wishes[i], wishes[j]]);
    }
  }
  
  // 随机选择rounds个不重复的配对
  const shuffledPairs = shuffleArray(allPairs);
  
  for (let i = 0; i < Math.min(rounds, shuffledPairs.length); i++) {
    // 随机决定左右位置
    const pair = Math.random() > 0.5 
      ? shuffledPairs[i] 
      : [shuffledPairs[i][1], shuffledPairs[i][0]];
    matchups.push(pair);
  }
  
  return matchups;
}

/**
 * 处理用户选择
 * @param {Object} winner 选中的愿望
 * @param {Object} loser 未选中的愿望
 * @param {Object} results 当前结果对象
 * @returns {Object} 更新后的结果对象
 */
function processChoice(winner, loser, results) {
  const newResults = { ...results };
  
  // 初始化胜者
  if (!newResults[winner.id]) {
    newResults[winner.id] = {
      id: winner.id,
      name: winner.name,
      wins: 0
    };
  }
  newResults[winner.id].wins += 1;
  
  // 确保败者也在结果中
  if (!newResults[loser.id]) {
    newResults[loser.id] = {
      id: loser.id,
      name: loser.name,
      wins: 0
    };
  }
  
  return newResults;
}

/**
 * 生成排名结果
 * @param {Object} results 结果对象 { wishId: { id, name, wins } }
 * @param {Array} allWishes 所有愿望列表(用于补全未参与的愿望)
 * @returns {Array} 排序后的愿望列表，包含rank字段
 */
function generateRanking(results, allWishes) {
  // 补全未参与对决的愿望
  const completeResults = { ...results };
  allWishes.forEach(wish => {
    if (!completeResults[wish.id]) {
      completeResults[wish.id] = {
        id: wish.id,
        name: wish.name,
        wins: 0
      };
    }
  });
  
  // 按胜场排序
  const sorted = Object.values(completeResults)
    .sort((a, b) => {
      // 胜场多的排前面
      if (b.wins !== a.wins) {
        return b.wins - a.wins;
      }
      // 胜场相同则随机
      return Math.random() - 0.5;
    })
    .map((item, index) => ({
      ...item,
      rank: index + 1
    }));
  
  return sorted;
}

/**
 * 分配到命运盘三圈
 * @param {Array} ranking 排名结果
 * @returns {Object} { core, main, later }
 */
function assignToDestinyWheel(ranking) {
  return {
    core: ranking.filter(w => w.rank === 1),      // 核心愿望(1个)
    main: ranking.filter(w => w.rank >= 2 && w.rank <= 3),  // 主线愿望(2个)
    later: ranking.filter(w => w.rank > 3)        // 以后再说(其余)
  };
}

/**
 * 交换核心愿望
 * @param {string} newCoreId 新核心愿望ID
 * @param {Object} wheel 当前命运盘 { core, main, later }
 * @param {Array} allWishes 所有愿望列表
 * @returns {Object} 更新后的命运盘
 */
function swapCoreWish(newCoreId, wheel, allWishes) {
  const oldCore = wheel.core[0];
  const newCore = allWishes.find(w => w.id === newCoreId);
  
  if (!newCore || newCore.id === oldCore.id) {
    return wheel;
  }
  
  const newWheel = {
    core: [{ ...newCore, rank: 1 }],
    main: [...wheel.main],
    later: [...wheel.later]
  };
  
  // 从main或later中移除新核心
  newWheel.main = newWheel.main.filter(w => w.id !== newCoreId);
  newWheel.later = newWheel.later.filter(w => w.id !== newCoreId);
  
  // 将旧核心放入main
  newWheel.main.unshift({ ...oldCore, rank: 2 });
  
  // 限制main最多3个
  if (newWheel.main.length > 3) {
    const overflow = newWheel.main.pop();
    newWheel.later.unshift(overflow);
  }
  
  return newWheel;
}

module.exports = {
  shuffleArray,
  generateMatchups,
  processChoice,
  generateRanking,
  assignToDestinyWheel,
  swapCoreWish
};
```

### 3.2 单元测试 `test/wish-sort.test.js`

```javascript
const {
  shuffleArray,
  generateMatchups,
  processChoice,
  generateRanking,
  assignToDestinyWheel,
  swapCoreWish
} = require('../utils/wish-sort.js');

// 测试数据
const mockWishes = [
  { id: 'health', name: '健康' },
  { id: 'wealth', name: '财富' },
  { id: 'growth', name: '成长' },
  { id: 'family', name: '家庭' },
  { id: 'freedom', name: '自由' }
];

describe('generateMatchups', () => {
  test('should generate correct number of matchups', () => {
    const matchups = generateMatchups(mockWishes, 7);
    expect(matchups.length).toBe(7);
  });
  
  test('each matchup should have 2 different wishes', () => {
    const matchups = generateMatchups(mockWishes, 5);
    matchups.forEach(pair => {
      expect(pair.length).toBe(2);
      expect(pair[0].id).not.toBe(pair[1].id);
    });
  });
});

describe('processChoice', () => {
  test('should increment winner wins', () => {
    const results = {};
    const winner = mockWishes[0];
    const loser = mockWishes[1];
    
    const newResults = processChoice(winner, loser, results);
    
    expect(newResults[winner.id].wins).toBe(1);
    expect(newResults[loser.id].wins).toBe(0);
  });
  
  test('should accumulate wins', () => {
    let results = {};
    const winner = mockWishes[0];
    
    results = processChoice(winner, mockWishes[1], results);
    results = processChoice(winner, mockWishes[2], results);
    
    expect(results[winner.id].wins).toBe(2);
  });
});

describe('generateRanking', () => {
  test('should rank by wins', () => {
    const results = {
      'health': { id: 'health', name: '健康', wins: 3 },
      'wealth': { id: 'wealth', name: '财富', wins: 1 },
      'growth': { id: 'growth', name: '成长', wins: 2 }
    };
    
    const ranking = generateRanking(results, mockWishes);
    
    expect(ranking[0].id).toBe('health');
    expect(ranking[0].rank).toBe(1);
  });
  
  test('should include all wishes', () => {
    const results = {
      'health': { id: 'health', name: '健康', wins: 1 }
    };
    
    const ranking = generateRanking(results, mockWishes);
    
    expect(ranking.length).toBe(mockWishes.length);
  });
});

describe('assignToDestinyWheel', () => {
  test('should assign top 1 to core', () => {
    const ranking = [
      { id: 'health', rank: 1 },
      { id: 'wealth', rank: 2 },
      { id: 'growth', rank: 3 },
      { id: 'family', rank: 4 }
    ];
    
    const wheel = assignToDestinyWheel(ranking);
    
    expect(wheel.core.length).toBe(1);
    expect(wheel.core[0].id).toBe('health');
    expect(wheel.main.length).toBe(2);
    expect(wheel.later.length).toBe(1);
  });
});

describe('swapCoreWish', () => {
  test('should swap core with main wish', () => {
    const wheel = {
      core: [{ id: 'health', name: '健康', rank: 1 }],
      main: [{ id: 'wealth', name: '财富', rank: 2 }],
      later: []
    };
    
    const newWheel = swapCoreWish('wealth', wheel, mockWishes);
    
    expect(newWheel.core[0].id).toBe('wealth');
    expect(newWheel.main.find(w => w.id === 'health')).toBeTruthy();
  });
});
```
