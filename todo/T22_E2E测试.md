### 4.1 小程序端测试脚本

```javascript
// test/e2e/full-flow.test.js

/**
 * E2E测试 - 首次用户完整流程
 */

const miniprogram = require('miniprogram-automator');

describe('首次用户完整流程', () => {
  let mini;
  let page;
  
  beforeAll(async () => {
    mini = await miniprogram.launch({
      projectPath: '/path/to/miniprogram',
      // 使用测试AppID
    });
  });
  
  afterAll(async () => {
    await mini.close();
  });
  
  beforeEach(async () => {
    // 清除Storage
    await mini.evaluate(() => {
      wx.clearStorageSync();
    });
  });
  
  test('TC-001: Onboarding到Daily完整流程', async () => {
    // 1. 进入Onboarding页
    page = await mini.reLaunch('/pages/onboarding/index');
    await page.waitFor(2000); // 等待动画
    
    // 验证Onboarding元素
    const dialogue = await page.$('.dialogue-bubble');
    expect(dialogue).not.toBeNull();
    
    // 点击开始按钮
    const startBtn = await page.$('.btn-primary');
    await startBtn.tap();
    await page.waitFor(500);
    
    // 2. Role页 - 选择天赋
    page = await mini.currentPage();
    expect(page.path).toBe('pages/role/index');
    
    const talentCards = await page.$$('.talent-card');
    expect(talentCards.length).toBe(4);
    
    // 选择第一个天赋
    await talentCards[0].tap();
    await page.waitFor(200);
    
    // 点击确认
    const confirmBtn = await page.$('.btn-primary');
    await confirmBtn.tap();
    await page.waitFor(500);
    
    // 3. Wishes页 - 完成7轮对决
    page = await mini.currentPage();
    expect(page.path).toBe('pages/wishes/index');
    
    for (let round = 0; round < 7; round++) {
      const wishCards = await page.$$('.battle-card');
      await wishCards[0].tap();
      await page.waitFor(400);
    }
    
    // 等待命运盘
    await page.waitFor(500);
    
    // 确认愿望
    const wishConfirmBtn = await page.$('.btn-primary');
    await wishConfirmBtn.tap();
    await page.waitFor(500);
    
    // 4. Workshop页 - 生成卡组
    page = await mini.currentPage();
    expect(page.path).toBe('pages/workshop/index');
    
    // 召唤灵感
    const summonBtn = await page.$('.action-btn');
    await summonBtn.tap();
    await page.waitFor(1000);
    
    // 选择一个行为
    const bubbles = await page.$$('.bubble-chip');
    if (bubbles.length > 0) {
      await bubbles[0].tap();
      await page.waitFor(500);
      
      // 选择锚点
      const anchorOptions = await page.$$('.anchor-option');
      if (anchorOptions.length > 0) {
        await anchorOptions[0].tap();
        await page.waitFor(300);
      }
    }
    
    // 生成卡组
    const generateBtn = await page.$('.btn-primary');
    await generateBtn.tap();
    await page.waitFor(2000);
    
    // 确认卡组
    const deckConfirmBtn = await page.$('.btn-primary');
    await deckConfirmBtn.tap();
    await page.waitFor(500);
    
    // 5. Daily页 - 完成任务
    page = await mini.currentPage();
    expect(page.path).toBe('pages/daily/index');
    
    // 点击开始
    const startDailyBtn = await page.$('.btn-primary');
    await startDailyBtn.tap();
    await page.waitFor(200);
    
    // 点击完成
    const completeBtn = await page.$('.btn-primary');
    await completeBtn.tap();
    await page.waitFor(500);
    
    // 击碎裂隙
    const shatterBtn = await page.$('.shatter-btn');
    await shatterBtn.tap();
    await page.waitFor(1000);
    
    // 验证完成
    const celebrationOverlay = await page.$('.celebration-overlay.visible');
    expect(celebrationOverlay).not.toBeNull();
    
    // 关闭庆祝
    const closeBtn = await page.$('.celebration-content .btn-primary');
    await closeBtn.tap();
    await page.waitFor(300);
    
    // 验证状态
    const state = await mini.evaluate(() => {
      return wx.getStorageSync('tinySpellsGameState');
    });
    
    expect(state.day).toBe(2);
    expect(state.streak).toBe(1);
    expect(state.todayDone).toBe(true);
  });
});
```

### 4.3 API测试

```javascript
// test/api/api.test.js

const request = require('supertest');
const app = require('../../server/src/app');

describe('API测试', () => {
  
  describe('GET /api/health', () => {
    it('应返回健康状态', async () => {
      const res = await request(app).get('/api/health');
      expect(res.status).toBe(200);
      expect(res.body.status).toBe('ok');
    });
  });
  
  describe('POST /api/ai/anchors', () => {
    it('应返回锚点建议', async () => {
      const res = await request(app)
        .post('/api/ai/anchors')
        .send({
          behavior: { name: '喝水', category: 'health' },
          context: { currentWish: '健康' }
        });
      
      expect(res.status).toBe(200);
      expect(res.body.success).toBe(true);
      expect(res.body.anchors.length).toBeGreaterThan(0);
    });
    
    it('缺少behavior应返回400', async () => {
      const res = await request(app)
        .post('/api/ai/anchors')
        .send({});
      
      expect(res.status).toBe(400);
    });
  });
  
  describe('POST /api/ai/map', () => {
    it('应返回MAP评估结果', async () => {
      const res = await request(app)
        .post('/api/ai/map')
        .send({
          deck: [
            { name: '喝水', type: 'base', seconds: 30, anchor: '起床后' },
            { name: '伸展', type: 'main', seconds: 60, anchor: '午休后' },
            { name: '深呼吸', type: 'bonus', seconds: 20, anchor: '睡前' }
          ]
        });
      
      expect(res.status).toBe(200);
      expect(res.body.success).toBe(true);
      expect(['Pass', 'Warning', 'Fail']).toContain(res.body.overall);
    });
  });
});
```
