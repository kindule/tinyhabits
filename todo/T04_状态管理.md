### 3.1 状态结构定义 `utils/state.js`

```javascript
/**
 * 游戏状态结构
 * 存储Key: 'tinySpellsGameState'
 */
const STATE_KEY = 'tinySpellsGameState';

/**
 * 获取初始状态
 */
function getInitialState() {
  return {
    // ===== 流程状态 =====
    day: 1,                     // 当前Day进度(1~7)
    hasRole: false,             // 是否完成角色创建
    hasWishRank: false,         // 是否完成愿望排序
    hasDeck: false,             // 是否生成卡组
    todayDone: false,           // 今日是否已完成
    lastDoneDate: null,         // 最后完成日期(String)
    
    // ===== 角色数据 =====
    roleTalent: null,           // 天赋类型: 'fire'|'tide'|'mason'|'wind'
    
    // ===== 愿望数据 =====
    wishRankResult: [],         // 愿望排序结果: [{id, name, wins, rank}]
    currentWish: null,          // 当前核心愿望名称
    currentWishId: null,        // 当前核心愿望ID
    
    // ===== 卡组数据 =====
    deck: [],                   // 3张卡: [{id, name, desc, seconds, difficulty, type, anchor, anchorEmoji, wish, createdAt}]
    materialArea: [],           // 素材区行为: [{id, name, anchor, anchorEmoji}]
    
    // ===== 游戏数据 =====
    streak: 0,                  // 当前连续天数
    bestStreak: 0,              // 最佳连续天数
    totalScore: 0,              // 总积分
    
    // ===== 收集数据 =====
    spellBook: [],              // 已解锁咒语图鉴: [咒语名称字符串]
    history: [],                // 最近7天完成记录: [{date, cardName, seconds}]
    
    // ===== 元数据 =====
    createdAt: new Date().toISOString(),
    version: '1.0'
  };
}

/**
 * 加载状态
 * @returns {Object} 游戏状态
 */
function loadState() {
  try {
    const data = wx.getStorageSync(STATE_KEY);
    if (!data) {
      return getInitialState();
    }
    // 合并默认值，防止版本升级时缺少字段
    return { ...getInitialState(), ...data };
  } catch (e) {
    console.error('加载状态失败:', e);
    return getInitialState();
  }
}

/**
 * 保存状态
 * @param {Object} state 游戏状态
 * @returns {boolean} 是否保存成功
 */
function saveState(state) {
  try {
    wx.setStorageSync(STATE_KEY, state);
    return true;
  } catch (e) {
    console.error('保存状态失败:', e);
    return false;
  }
}

/**
 * 清空状态(用于调试/重置)
 * @returns {boolean} 是否清空成功
 */
function clearState() {
  try {
    wx.removeStorageSync(STATE_KEY);
    return true;
  } catch (e) {
    console.error('清空状态失败:', e);
    return false;
  }
}

/**
 * 更新部分状态
 * @param {Object} updates 需要更新的字段
 * @returns {Object} 更新后的完整状态
 */
function updateState(updates) {
  const state = loadState();
  const newState = { ...state, ...updates };
  saveState(newState);
  return newState;
}

module.exports = {
  STATE_KEY,
  getInitialState,
  loadState,
  saveState,
  clearState,
  updateState
};
```

### 3.2 日期工具函数 `utils/date.js`

```javascript
/**
 * 判断是否是今天
 * @param {string} dateString 日期字符串
 * @returns {boolean}
 */
function isToday(dateString) {
  if (!dateString) return false;
  const date = new Date(dateString);
  const today = new Date();
  return date.toDateString() === today.toDateString();
}

/**
 * 判断是否是昨天
 * @param {string} dateString 日期字符串
 * @returns {boolean}
 */
function isYesterday(dateString) {
  if (!dateString) return false;
  const date = new Date(dateString);
  const yesterday = new Date();
  yesterday.setDate(yesterday.getDate() - 1);
  return date.toDateString() === yesterday.toDateString();
}

/**
 * 获取今天的日期字符串
 * @returns {string}
 */
function getTodayString() {
  return new Date().toDateString();
}

/**
 * 判断是否需要重置todayDone
 * @param {Object} state 游戏状态
 * @returns {boolean}
 */
function shouldResetTodayDone(state) {
  if (!state.lastDoneDate) return false;
  return !isToday(state.lastDoneDate);
}

module.exports = {
  isToday,
  isYesterday,
  getTodayString,
  shouldResetTodayDone
};
```

### 3.3 流程验证函数 `utils/flow.js`

```javascript
const { loadState, saveState } = require('./state.js');
const { shouldResetTodayDone } = require('./date.js');

/**
 * 验证流程，返回需要跳转的页面
 * @returns {Object} { redirect: string | null }
 */
function validateFlow() {
  const state = loadState();
  
  if (!state.hasRole) {
    return { redirect: '/pages/role/index' };
  }
  if (!state.hasWishRank) {
    return { redirect: '/pages/wishes/index' };
  }
  if (!state.hasDeck) {
    return { redirect: '/pages/workshop/index' };
  }
  
  return { redirect: null };
}

/**
 * 刷新每日状态(App onShow时调用)
 */
function refreshDailyState() {
  const state = loadState();
  
  if (shouldResetTodayDone(state)) {
    state.todayDone = false;
    saveState(state);
  }
  
  return state;
}

/**
 * 获取流程进度
 * @returns {Object} 各阶段完成状态
 */
function getFlowProgress() {
  const state = loadState();
  return {
    hasRole: state.hasRole,
    hasWishRank: state.hasWishRank,
    hasDeck: state.hasDeck,
    day: state.day,
    todayDone: state.todayDone
  };
}

module.exports = {
  validateFlow,
  refreshDailyState,
  getFlowProgress
};
```

### 3.4 Streak计算模块 `utils/streak.js`

```javascript
const { loadState, saveState } = require('./state.js');
const { isToday, isYesterday, getTodayString } = require('./date.js');

/**
 * 更新Streak
 * @param {Object} state 游戏状态
 * @returns {number} 更新后的Streak值
 */
function updateStreak(state) {
  const lastDoneDate = state.lastDoneDate;
  
  if (!lastDoneDate) {
    // 首次完成
    state.streak = 1;
  } else if (isYesterday(lastDoneDate)) {
    // 昨天完成: 连续+1
    state.streak += 1;
  } else if (!isToday(lastDoneDate)) {
    // 超过1天: 重置
    state.streak = 1;
  }
  // 今天已完成的情况不应进入此函数
  
  // 更新最佳Streak
  if (state.streak > state.bestStreak) {
    state.bestStreak = state.streak;
  }
  
  return state.streak;
}

/**
 * 检查Streak状态
 * @returns {Object} { status: string, message: string | null }
 */
function checkStreakStatus() {
  const state = loadState();
  
  if (!state.lastDoneDate) {
    return { status: 'new', message: null };
  }
  
  if (isToday(state.lastDoneDate)) {
    return { status: 'completed', message: '今日已完成' };
  }
  
  if (isYesterday(state.lastDoneDate)) {
    return { status: 'continuing', message: null };
  }
  
  // 超过1天未完成
  return { 
    status: 'broken', 
    message: '救援咒语已准备好,微光不会消失' 
  };
}

module.exports = {
  updateStreak,
  checkStreakStatus
};
```

describe('State Management', () => {
  // 测试初始化状态
