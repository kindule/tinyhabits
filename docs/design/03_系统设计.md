```javascript
// 生成对决配对
function generateMatchups(wishes, rounds = 7) {
  const matchups = [];
  const used = new Set();
  
  for (let i = 0; i < rounds; i++) {
    // 随机选择两个不同的愿望
    let pair;
    do {
      const shuffled = shuffleArray([...wishes]);
      pair = [shuffled[0], shuffled[1]];
    } while (used.has(`${pair[0].id}-${pair[1].id}`));
    
    used.add(`${pair[0].id}-${pair[1].id}`);
    used.add(`${pair[1].id}-${pair[0].id}`);
    matchups.push(pair);
  }
  
  return matchups;
}

// 处理用户选择
function handleChoice(winner, loser, results) {
  // 增加胜者的胜场数
  if (!results[winner.id]) {
    results[winner.id] = { wins: 0, id: winner.id, name: winner.name };
  }
  results[winner.id].wins += 1;
  
  // 确保败者也在结果中
  if (!results[loser.id]) {
    results[loser.id] = { wins: 0, id: loser.id, name: loser.name };
  }
}

// 生成排序结果
function generateRanking(results) {
  const sorted = Object.values(results)
    .sort((a, b) => b.wins - a.wins)
    .map((item, index) => ({
      ...item,
      rank: index + 1
    }));
  
  return sorted;
}
```

function assignToDestinyWheel(ranking) {

    main: ranking.filter(w => w.rank >= 2 && w.rank <= 3),  // 主线愿望(2个)

// 用户调整核心愿望
function swapCoreWish(newCoreId, wheel) {
  const oldCore = wheel.core[0];
  const newCore = [...wheel.main, ...wheel.later].find(w => w.id === newCoreId);
  
  if (newCore) {
    // 交换位置
    if (wheel.main.find(w => w.id === newCoreId)) {
      wheel.main = wheel.main.filter(w => w.id !== newCoreId);
      wheel.main.push(oldCore);
    } else {
      wheel.later = wheel.later.filter(w => w.id !== newCoreId);
      wheel.later.push(oldCore);
    }
    wheel.core = [newCore];
  }
  
  return wheel;
}

function generateDeck(materialArea, currentWish) {
  // 按难度/时长分类
  const easy = materialArea.filter(b => b.seconds <= 60 && b.difficulty === 1);
  const medium = materialArea.filter(b => b.seconds > 60 && b.seconds <= 300);
  const fun = materialArea.filter(b => b.seconds <= 30);
  
  // 生成三张卡
  const deck = [];
  
  // 保底卡: 最简单的
  if (easy.length > 0) {
    const base = easy[Math.floor(Math.random() * easy.length)];
    deck.push(createCard(base, 'base', currentWish));
  } else {
    deck.push(createDefaultBaseCard(currentWish));
  }
  
  // 主线卡: 中等难度
  if (medium.length > 0) {
    const main = medium[Math.floor(Math.random() * medium.length)];
    deck.push(createCard(main, 'main', currentWish));
  } else {
    deck.push(createDefaultMainCard(currentWish));
  }
  
  // 彩蛋卡: 趣味性
  if (fun.length > 0) {
    const bonus = fun[Math.floor(Math.random() * fun.length)];
    deck.push(createCard(bonus, 'bonus', currentWish));
  } else {
    deck.push(createDefaultBonusCard(currentWish));
  }
  
  return deck;
}

function drawTodayCard(deck, streak) {
  // 连续天数不足2天: 必定保底卡
  if (streak < 2) {
    return deck.find(card => card.type === 'base') || deck[0];

  const rand = Math.random();

}

### 4.2 完成流程逻辑

```javascript
// 完成微行动
function completeHabit(state, cardId) {
  const today = new Date().toDateString();
  
  // 1. 幂等性检查
  if (state.todayDone && state.lastDoneDate === today) {
    return { success: false, message: '今日已完成' };
  }
  
  // 2. 更新Streak
  const streakResult = updateStreak(state);
  
  // 3. 更新Day进度
  if (state.day < 7) {
    state.day += 1;
  }
  
  // 4. 更新完成状态
  state.todayDone = true;
  state.lastDoneDate = today;
  
  // 5. 更新积分
  state.totalScore += 10;
  
  // 6. 记录历史
  const card = state.deck.find(c => c.id === cardId);
  state.history.push({
    date: today,
    cardName: card ? card.name : '未知咒语',
    seconds: card ? card.seconds : 0
  });
  
  // 保持历史记录最多7条
  if (state.history.length > 7) {
    state.history = state.history.slice(-7);
  }
  
  // 7. 更新咒语图鉴
  if (card && !state.spellBook.includes(card.name)) {
    state.spellBook.push(card.name);
  }
  
  saveState(state);
  
  return { 
    success: true, 
    streak: state.streak,
    day: state.day,
    score: state.totalScore,
    isDay7: state.day === 7
  };
}

// 更新Streak
function updateStreak(state) {
  const today = new Date();
  const lastDate = state.lastDoneDate ? new Date(state.lastDoneDate) : null;
  
  if (!lastDate) {
    // 首次完成
    state.streak = 1;
  } else {
    const yesterday = new Date(today);
    yesterday.setDate(yesterday.getDate() - 1);
    
    if (lastDate.toDateString() === yesterday.toDateString()) {
      // 昨天完成: 连续+1
      state.streak += 1;
    } else if (lastDate.toDateString() !== today.toDateString()) {
      // 超过1天: 重置
      state.streak = 1;
    }
    // 今天已完成的情况在幂等性检查中处理
  }
  
  // 更新最佳Streak
  if (state.streak > state.bestStreak) {
    state.bestStreak = state.streak;
  }
  
  return state.streak;
}
```

// 判断是否是今天
function isToday(dateString) {
  if (!dateString) return false;
  const date = new Date(dateString);
  const today = new Date();
  return date.toDateString() === today.toDateString();
}

// 判断是否是昨天
function isYesterday(dateString) {
  if (!dateString) return false;
  const date = new Date(dateString);
  const yesterday = new Date();
  yesterday.setDate(yesterday.getDate() - 1);
  return date.toDateString() === yesterday.toDateString();
}

// 判断是否需要重置todayDone
function shouldResetTodayDone(state) {
  if (!state.lastDoneDate) return false;
  return !isToday(state.lastDoneDate);
}

// 每日状态刷新(App onShow时调用)
function refreshDailyState(state) {
  if (shouldResetTodayDone(state)) {
    state.todayDone = false;
    saveState(state);
  }
  return state;
}

  try {
    return wx.getStorageSync(STORAGE_KEY);
  } catch (e) {
    console.error('Storage load error:', e);
    return null;
  }
}

// 写入
function save(data) {
  try {
    wx.setStorageSync(STORAGE_KEY, data);
    return true;
  } catch (e) {
    console.error('Storage save error:', e);
    return false;
  }
}

// 清空
function clear() {
  try {
    wx.removeStorageSync(STORAGE_KEY);
    return true;
  } catch (e) {
    console.error('Storage clear error:', e);
    return false;
  }
}

// 获取存储信息
function getInfo() {
  try {
    const data = wx.getStorageSync(STORAGE_KEY);
    return {
      exists: !!data,
      size: JSON.stringify(data || {}).length
    };
  } catch (e) {
    return { exists: false, size: 0 };
  }
}

### 7.1 异常处理策略

```javascript
// 全局错误处理
const ErrorHandler = {
  // 网络错误
  handleNetworkError(error) {
    console.error('Network error:', error);
    return {
      showToast: true,
      message: '网络不太好,请稍后重试',
      fallback: true  // 使用降级方案
    };
  },
  
  // 存储错误
  handleStorageError(error) {
    console.error('Storage error:', error);
    return {
      showToast: true,
      message: '数据保存失败,请重试',
      fallback: false
    };
  },
  
  // AI服务错误
  handleAIError(error) {
    console.error('AI error:', error);
    return {
      showToast: false,
      message: null,
      fallback: true  // 使用预设库
    };
  },
  
  // 状态异常
  handleStateError(error) {
    console.error('State error:', error);
    // 尝试恢复初始状态
    return {
      showToast: true,
      message: '状态异常,正在恢复...',
      action: 'reset'
    };
  }
};
```
