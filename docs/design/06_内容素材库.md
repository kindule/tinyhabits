## 七、合成模板

### 7.1 咒语命名模板

```javascript
const MERGE_TEMPLATES = [
  '{A}后{B}',           // 喝水后伸展
  '{A}+{B}',            // 深呼吸+喝水
  '{A}·{B}',            // 水润·舒展
  '「{A}&{B}」',         // 「水&伸展」
  '{A}时{B}',           // 站起来时看远处
  '先{A}再{B}'          // 先喝水再伸展
];

// 生成合成名称
function generateMergeName(behaviorA, behaviorB) {
  const template = MERGE_TEMPLATES[Math.floor(Math.random() * MERGE_TEMPLATES.length)];
  
  // 简化行为名称(去掉动词前缀)
  const nameA = behaviorA.name.replace(/^(去|做|来|打开|点击)/, '');
  const nameB = behaviorB.name.replace(/^(去|做|来|打开|点击)/, '');
  
  return template
    .replace('{A}', nameA)
    .replace('{B}', nameB);
}

module.exports = { MERGE_TEMPLATES, generateMergeName };
```

### 9.2 使用方式

```javascript
// 在页面中使用
const { WISH_CARDS } = require('../../utils/wishes.js');
const { BEHAVIORS } = require('../../utils/behaviors.js');
const { ARC_ENCOURAGEMENTS } = require('../../utils/dialogues.js');

// 随机获取鼓励台词
function getRandomEncouragement() {
  return ARC_ENCOURAGEMENTS[Math.floor(Math.random() * ARC_ENCOURAGEMENTS.length)];
}

// 获取分类行为
function getBehaviorsByCategory(category) {
  return BEHAVIORS.filter(b => b.category === category);
}
```
